cmake_minimum_required(VERSION 3.21)
project(omarchy-ime-fcitx5 VERSION 0.1.0 LANGUAGES CXX)
find_package(Fcitx5Core REQUIRED)
find_library(OMARCHY_IME_RUST NAMES omarchy_ime PATHS ${CMAKE_CURRENT_SOURCE_DIR}/../../target/release REQUIRED)
add_library(omarchy_ime_rust SHARED IMPORTED)
set_target_properties(omarchy_ime_rust PROPERTIES IMPORTED_LOCATION "${OMARCHY_IME_RUST}")
add_library(omarchy-ime-fcitx5 SHARED src/omarchy-ime.cpp)
target_compile_features(omarchy-ime-fcitx5 PRIVATE cxx_std_17)
target_link_libraries(omarchy-ime-fcitx5 PRIVATE Fcitx5::Core omarchy_ime_rust)
set_target_properties(omarchy-ime-fcitx5 PROPERTIES PREFIX "" INSTALL_RPATH "$ORIGIN")
install(TARGETS omarchy-ime-fcitx5 LIBRARY DESTINATION lib/fcitx5)
install(FILES data/omarchy-ime-addon.conf DESTINATION share/fcitx5/addon RENAME omarchy-ime.conf)
install(FILES data/omarchy-ime.conf DESTINATION share/fcitx5/inputmethod RENAME omarchy-ime.conf)
install(FILES data/omarchy-ime-wubi.conf DESTINATION share/fcitx5/inputmethod)
